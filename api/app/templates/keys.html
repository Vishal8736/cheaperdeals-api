import httpx, time
def ping(url):                      # api/app/services/keepalive.py mein yahi hai
    t = time.monotonic()
    r = httpx.get(url.rstrip("/") + "/ping", timeout=15)
    return r.status_code == 200, int((time.monotonic() - t) * 1000)
